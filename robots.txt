# Post-signup pages. Nothing secret, but they only make sense mid-flow and
# would be dead ends in a search result. All three also send noindex in the
# markup, which is what actually removes them from an index.
Disallow: /checkout.html
Disallow: /welcome.html
Disallow: /dashboard.html
Disallow: /api/

# Identity fetches these at send time to build the transactional mail. They are
# templates, not pages — a search result full of unsubstituted {{ … }} helps
# nobody.
Disallow: /email-templates/

# Maintainer notes. netlify.toml already returns 404 for these; this keeps a
# crawler from asking in the first place.
Disallow: /AGENTS.md
Disallow: /CLAUDE.local.md
Disallow: /README.md
